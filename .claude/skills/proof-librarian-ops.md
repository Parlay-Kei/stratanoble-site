Manage proof receipts and audit trails. Archive completed work with evidence. Enable traceability and compliance.

## Level 1: Receipt Management

### storeReceipt()
```javascript
/**
 * Store a proof receipt
 */
async function storeReceipt(receipt) {
  const fs = require('fs').promises;
  const path = require('path');
  const crypto = require('crypto');

  const timestamp = new Date().toISOString();
  const date = timestamp.split('T')[0];
  const year = date.split('-')[0];
  const month = date.substring(0, 7);

  // Generate receipt ID
  const receiptId = `RCPT-${receipt.type || 'GEN'}-${timestamp.replace(/[:.]/g, '-')}`;

  // Calculate hash for integrity
  const hash = crypto.createHash('sha256')
    .update(JSON.stringify(receipt))
    .digest('hex');

  // Build receipt document
  const doc = {
    id: receiptId,
    hash,
    timestamp,
    ...receipt,
    metadata: {
      version: '1.0',
      schema: 'anx-proof-receipt',
      generatedBy: 'proof-librarian-ops'
    }
  };

  // Store in appropriate directory
  const proofDir = `C:/Dev/.claude-anx/docs/ops/04-PROOFS/${year}/${month}`;
  await fs.mkdir(proofDir, { recursive: true });

  const receiptPath = path.join(proofDir, `${receiptId}.md`);
  const receiptContent = formatReceiptMarkdown(doc);

  await fs.writeFile(receiptPath, receiptContent, 'utf-8');

  // Also store JSON for machine reading
  const jsonPath = path.join(proofDir, `${receiptId}.json`);
  await fs.writeFile(jsonPath, JSON.stringify(doc, null, 2), 'utf-8');

  // Update index
  await updateProofIndex(doc);

  return {
    receiptId,
    path: receiptPath,
    hash,
    timestamp
  };
}
```

## Level 2: Search and Retrieval

### searchProofs()
```javascript
/**
 * Search proofs by criteria
 */
async function searchProofs(criteria = {}) {
  const fs = require('fs').promises;
  const path = require('path');

  const indexPath = 'C:/Dev/.claude-anx/docs/ops/04-PROOFS/index.json';

  try {
    const content = await fs.readFile(indexPath, 'utf-8');
    let index = JSON.parse(content);

    // Apply filters
    if (criteria.type) {
      index = index.filter(p => p.type === criteria.type);
    }
    if (criteria.agent) {
      index = index.filter(p => p.agent === criteria.agent);
    }
    if (criteria.ticketId) {
      index = index.filter(p => p.ticketId === criteria.ticketId);
    }
    if (criteria.from) {
      const fromDate = new Date(criteria.from);
      index = index.filter(p => new Date(p.timestamp) >= fromDate);
    }
    if (criteria.to) {
      const toDate = new Date(criteria.to);
      index = index.filter(p => new Date(p.timestamp) <= toDate);
    }
    if (criteria.keyword) {
      const kw = criteria.keyword.toLowerCase();
      index = index.filter(p =>
        p.summary?.toLowerCase().includes(kw) ||
        p.id.toLowerCase().includes(kw)
      );
    }

    // Sort by timestamp descending
    index.sort((a, b) => new Date(b.timestamp) - new Date(a.timestamp));

    // Limit results
    const limit = criteria.limit || 50;
    index = index.slice(0, limit);

    return {
      count: index.length,
      results: index
    };
  } catch (err) {
    return { count: 0, results: [], error: err.message };
  }
}

/**
 * Retrieve specific proof
 */
async function retrieveProof(receiptId) {
  const fs = require('fs').promises;
  const path = require('path');

  // Parse date from receipt ID
  const match = receiptId.match(/RCPT-\w+-(\d{4})-(\d{2})-(\d{2})/);
  if (!match) {
    throw new Error(`Invalid receipt ID format: ${receiptId}`);
  }

  const [, year, month] = match;
  const proofDir = `C:/Dev/.claude-anx/docs/ops/04-PROOFS/${year}/${year}-${month}`;

  const jsonPath = path.join(proofDir, `${receiptId}.json`);
  const mdPath = path.join(proofDir, `${receiptId}.md`);

  try {
    const jsonContent = await fs.readFile(jsonPath, 'utf-8');
    const mdContent = await fs.readFile(mdPath, 'utf-8');

    return {
      found: true,
      data: JSON.parse(jsonContent),
      markdown: mdContent,
      paths: { json: jsonPath, md: mdPath }
    };
  } catch (err) {
    return { found: false, error: err.message };
  }
}
```

### createProofPack()
```javascript
/**
 * Create a proof pack for a ticket/milestone
 */
async function createProofPack(config) {
  const {
    ticketId,
    title,
    summary,
    receipts = [],
    criteria = {}
  } = config;

  // If criteria provided, search for receipts
  let packReceipts = receipts;
  if (Object.keys(criteria).length > 0) {
    const search = await searchProofs({ ...criteria, ticketId });
    packReceipts = search.results;
  }

  // Build proof pack
  const timestamp = new Date().toISOString();
  const packId = `PACK-${ticketId || 'GEN'}-${timestamp.replace(/[:.]/g, '-')}`;

  const pack = {
    id: packId,
    ticketId,
    title,
    summary,
    timestamp,
    receipts: packReceipts.map(r => ({
      id: r.id,
      type: r.type,
      timestamp: r.timestamp,
      summary: r.summary
    })),
    statistics: {
      totalReceipts: packReceipts.length,
      byType: countByType(packReceipts),
      timeSpan: calculateTimeSpan(packReceipts)
    }
  };

  // Generate pack document
  const packContent = formatProofPackMarkdown(pack);

  // Store pack
  const fs = require('fs').promises;
  const date = timestamp.split('T')[0];
  const year = date.split('-')[0];
  const month = date.substring(0, 7);

  const packDir = `C:/Dev/.claude-anx/docs/ops/04-PROOFS/${year}/${month}`;
  const packPath = `${packDir}/${packId}.md`;

  await fs.writeFile(packPath, packContent, 'utf-8');

  return {
    packId,
    path: packPath,
    receiptCount: packReceipts.length,
    timestamp
  };
}

/**
 * Format proof pack as markdown
 */
function formatProofPackMarkdown(pack) {
  return `# Proof Pack

**ID**: ${pack.id}
**Ticket**: ${pack.ticketId || 'N/A'}
**Title**: ${pack.title}
**Generated**: ${pack.timestamp}

---

## Summary

${pack.summary}

## Statistics

| Metric | Value |
|--------|-------|
| Total Receipts | ${pack.statistics.totalReceipts} |
| Time Span | ${pack.statistics.timeSpan} |
| Types | ${Object.entries(pack.statistics.byType).map(([k, v]) => `${k}: ${v}`).join(', ')} |

## Receipts

| ID | Type | Timestamp | Summary |
|----|------|-----------|---------|
${pack.receipts.map(r => `| ${r.id} | ${r.type} | ${r.timestamp} | ${r.summary || '-'} |`).join('\n')}

---

## Verification Checklist

- [ ] All receipts present and accessible
- [ ] Hash integrity verified
- [ ] Timeline consistent
- [ ] No gaps in evidence chain

---

*Generated by proof-librarian-ops v1.0*
`;
}

function countByType(receipts) {
  const counts = {};
  for (const r of receipts) {
    const type = r.type || 'unknown';
    counts[type] = (counts[type] || 0) + 1;
  }
  return counts;
}

function calculateTimeSpan(receipts) {
  if (receipts.length === 0) return 'N/A';
  const timestamps = receipts.map(r => new Date(r.timestamp)).sort((a, b) => a - b);
  const first = timestamps[0];
  const last = timestamps[timestamps.length - 1];
  const diffMs = last - first;
  const diffDays = Math.ceil(diffMs / (1000 * 60 * 60 * 24));
  return `${diffDays} days`;
}
```

## Level 4: Verification and Archival

### verifyProof()
```javascript
/**
 * Verify proof integrity
 */
async function verifyProof(receiptId) {
  const crypto = require('crypto');

  const proof = await retrieveProof(receiptId);
  if (!proof.found) {
    return { valid: false, error: 'Proof not found' };
  }

  // Recalculate hash
  const data = { ...proof.data };
  const storedHash = data.hash;
  delete data.hash;

  const calculatedHash = crypto.createHash('sha256')
    .update(JSON.stringify(data))
    .digest('hex');

  const valid = storedHash === calculatedHash;

  return {
    valid,
    receiptId,
    storedHash,
    calculatedHash,
    timestamp: proof.data.timestamp,
    message: valid ? 'Proof integrity verified' : 'INTEGRITY FAILURE - Hash mismatch'
  };
}

/**
 * Archive proofs to cold storage
 */
async function archiveProofs(criteria) {
  const fs = require('fs').promises;
  const path = require('path');

  // Find proofs matching criteria
  const search = await searchProofs(criteria);

  if (search.count === 0) {
    return { archived: 0, message: 'No proofs match criteria' };
  }

  // Create archive
  const timestamp = new Date().toISOString();
  const archiveId = `ARCHIVE-${timestamp.replace(/[:.]/g, '-')}`;
  const archiveDir = `C:/Dev/.claude-anx/docs/ops/04-PROOFS/archive`;
  await fs.mkdir(archiveDir, { recursive: true });

  const archive = {
    id: archiveId,
    timestamp,
    criteria,
    proofs: search.results
  };

  const archivePath = path.join(archiveDir, `${archiveId}.json`);
  await fs.writeFile(archivePath, JSON.stringify(archive, null, 2), 'utf-8');

  return {
    archived: search.count,
    archiveId,
    path: archivePath
  };
}
```
